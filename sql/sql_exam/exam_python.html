SQL — Create table & insert data (run once) 
CREATE TABLE world_cups (
    year                    INT PRIMARY KEY,
    host_country            VARCHAR(50) NOT NULL,
    winner                  VARCHAR(50) NOT NULL,
    start_date              DATE        NOT NULL,
    participating_countries INT         NOT NULL,
    total_goals             INT         NOT NULL,
    total_red_cards         INT,
    total_yellow_cards      INT,
    penalty_shootout        BOOLEAN     NOT NULL
);
