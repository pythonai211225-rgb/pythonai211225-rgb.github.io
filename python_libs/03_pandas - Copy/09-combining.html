python
valid_depts = pd.DataFrame({'dept_id': [10, 20]})

# keep only employees whose dept_id is in valid_depts
matched = employees[employees.dept_id.isin(valid_depts.dept_id)]
print(matched)
#    id   name  dept_id
# 0   1  Alice       10
# 1   2    Bob       10
# 2   3  Carol       20

# keep only employees whose dept_id is NOT in valid_depts (~ = NOT)
unmatched = employees[~employees.dept_id.isin(valid_depts.dept_id)]
print(unmatched)
#    id   name  dept_id
# 3   4  David       30

python
ydf = pd.DataFrame({'x1':['A','B','C'], 'x2':[1,2,3]})
zdf = pd.DataFrame({'x1':['B','C','D'], 'x2':[2,3,4]})

# intersection — rows in BOTH
print(pd.merge(ydf, zdf))
#   x1  x2
# 0  B   2
# 1  C   3

# union — rows in EITHER or both
print(pd.merge(ydf, zdf, how='outer'))

# set difference — rows in ydf but NOT in zdf
diff = (pd.merge(ydf, zdf, how='outer', indicator=True)
          .query('_merge == "left_only"')
          .drop(columns=['_merge']))
print(diff)
#   x1  x2
# 0  A   1
Exercises

Practice — Combining Data Sets

Q1

Perform an inner join on the two DataFrames below on the product_id key. How many rows does the result have?

orders   = pd.DataFrame({'order_id':[1,2,3], 'product_id':[101,102,105], 'qty':[2,1,3]})
products = pd.DataFrame({'product_id':[101,102,103], 'name':['Apple','Banana','Cherry']})
Show answer 
result = pd.merge(orders, products, how='inner', on='product_id')
print(result)
#    order_id  product_id  qty    name
# 0         1         101    2   Apple
# 1         2         102    1  Banana
# 2 rows — order 3 (product 105) has no match
Q2

Do a left join of the same DataFrames. What happens to order 3?

Show answer 
result = pd.merge(orders, products, how='left', on='product_id')
print(result)
#    order_id  product_id  qty    name
# 0         1         101    2   Apple
# 1         2         102    1  Banana
# 2         3         105    3     NaN  ← kept but name is NaN
Q3

Use isin() to find all orders whose product_id is NOT in the products table.

Show answer 
unmatched = orders[~orders.product_id.isin(products.product_id)]
print(unmatched)
#    order_id  product_id  qty
# 2         3         105    3
